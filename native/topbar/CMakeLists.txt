cmake_minimum_required(VERSION 3.21)
project(LumenTopbar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Quick QuickControls2 DBus)
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DBUSMENU REQUIRED IMPORTED_TARGET dbusmenu-glib-0.4 gio-2.0)
add_library(topbar-menu STATIC appmenu.cpp appmenu.h)
target_link_libraries(topbar-menu PUBLIC Qt6::Widgets Qt6::Quick Qt6::DBus PkgConfig::DBUSMENU)
target_compile_options(topbar-menu PRIVATE -Wall -Wextra -Werror)
add_library(topbar-tray STATIC tray.cpp tray.h)
target_link_libraries(topbar-tray PUBLIC topbar-menu Qt6::DBus)
target_compile_options(topbar-tray PRIVATE -Wall -Wextra -Werror)

get_filename_component(LUMEN_ICON_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../icons/hicolor/scalable/apps" ABSOLUTE)
file(GLOB LUMEN_ICONS CONFIGURE_DEPENDS "${LUMEN_ICON_ROOT}/lumen*.svg")
function(topbar_ui target)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(${target} PRIVATE topbar-menu topbar-tray Qt6::QuickControls2 LayerShellQt::Interface)
    qt_add_resources(${target} "${target}-qml" PREFIX "/lumen" FILES Topbar.qml BarButton.qml ControlSlider.qml SystemMenu.qml ActivityMenu.qml WindowState.qml StartMenu.qml AskComposer.qml ConversationState.qml ConversationPane.qml)
    set(SEARCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../files/qml/search")
    qt_add_resources(${target} "${target}-conversation" PREFIX "/lumen/search" BASE "${SEARCH_ROOT}" FILES "${SEARCH_ROOT}/AnswerBlocks.qml" "${SEARCH_ROOT}/ResultView.qml" "${SEARCH_ROOT}/QuestionCard.qml" "${SEARCH_ROOT}/LumenButton.qml")
    qt_add_resources(${target} "${target}-symbols" PREFIX "/lumen" FILES symbols/halo.svg symbols/lumen-mark.svg symbols/lumen-files.svg symbols/lumen-wissen.svg symbols/lumen-workbench.svg symbols/lumen-inbox.svg)
    qt_add_resources(${target} "${target}-icons" PREFIX "/lumen/icons" BASE "${LUMEN_ICON_ROOT}" FILES ${LUMEN_ICONS})
endfunction()
add_executable(lumen-topbar main.cpp paneldata.cpp paneldata.h conversation.cpp conversation.h)
topbar_ui(lumen-topbar)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    foreach(kind menu tray)
        add_executable(topbar-${kind}-test ${kind}_test.cpp)
        target_link_libraries(topbar-${kind}-test PRIVATE topbar-${kind} Qt6::Test)
        target_compile_options(topbar-${kind}-test PRIVATE -Wall -Wextra -Werror)
        add_test(NAME topbar-${kind} COMMAND dbus-run-session -- $<TARGET_FILE:topbar-${kind}-test>)
        set_tests_properties(topbar-${kind} PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endforeach()
    add_executable(topbar-data-test paneldata_test.cpp paneldata.cpp paneldata.h conversation.cpp conversation.h)
    target_link_libraries(topbar-data-test PRIVATE Qt6::Widgets Qt6::Quick Qt6::DBus Qt6::Test LayerShellQt::Interface)
    target_compile_options(topbar-data-test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME topbar-data COMMAND dbus-run-session -- $<TARGET_FILE:topbar-data-test>)
    set_tests_properties(topbar-data PROPERTIES TIMEOUT 10 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(topbar-ui-test ui_test.cpp paneldata.cpp paneldata.h conversation.cpp conversation.h)
    topbar_ui(topbar-ui-test)
    target_link_libraries(topbar-ui-test PRIVATE Qt6::Test)

    enable_language(C)
    find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
    pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
    set(INPUT_TEST "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/desktop")
    add_custom_command(OUTPUT fake-input-client.h fake-input-protocol.c
        COMMAND "${WAYLAND_SCANNER}" client-header "${INPUT_TEST}/fake-input.xml" fake-input-client.h
        COMMAND "${WAYLAND_SCANNER}" private-code "${INPUT_TEST}/fake-input.xml" fake-input-protocol.c
        DEPENDS "${INPUT_TEST}/fake-input.xml" VERBATIM)
    add_executable(topbar-input "${INPUT_TEST}/input.c" "${CMAKE_CURRENT_BINARY_DIR}/fake-input-protocol.c")
    set_target_properties(topbar-input PROPERTIES AUTOMOC OFF AUTORCC OFF)
    target_include_directories(topbar-input PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(topbar-input PRIVATE PkgConfig::WAYLAND)
endif()
