cmake_minimum_required(VERSION 3.21)
project(LumenTopbar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Quick QuickControls2 DBus Test)
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DBUSMENU REQUIRED IMPORTED_TARGET dbusmenu-glib-0.4 gio-2.0)
add_library(topbar-menu STATIC appmenu.cpp appmenu.h)
target_link_libraries(topbar-menu PUBLIC Qt6::Widgets Qt6::Quick PkgConfig::DBUSMENU)
target_compile_options(topbar-menu PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(topbar-menu-test appmenu_test.cpp)
target_link_libraries(topbar-menu-test PRIVATE topbar-menu Qt6::Test)
target_compile_options(topbar-menu-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME topbar-menu COMMAND dbus-run-session -- $<TARGET_FILE:topbar-menu-test>)
set_tests_properties(topbar-menu PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(lumen-topbar main.cpp paneldata.cpp paneldata.h)
target_link_libraries(lumen-topbar PRIVATE topbar-menu Qt6::QuickControls2 Qt6::DBus LayerShellQt::Interface)
target_compile_options(lumen-topbar PRIVATE -Wall -Wextra -Werror)
qt_add_resources(lumen-topbar "topbar-qml" PREFIX "/lumen" FILES Topbar.qml BarButton.qml WindowState.qml)
add_executable(topbar-ui-test ui_test.cpp paneldata.cpp paneldata.h)
target_link_libraries(topbar-ui-test PRIVATE topbar-menu Qt6::QuickControls2 Qt6::DBus Qt6::Test LayerShellQt::Interface)
target_compile_options(topbar-ui-test PRIVATE -Wall -Wextra -Werror)
qt_add_resources(topbar-ui-test "topbar-test-qml" PREFIX "/lumen" FILES Topbar.qml BarButton.qml WindowState.qml)
