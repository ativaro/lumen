cmake_minimum_required(VERSION 3.21)
project(lumen_vault_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBSSH REQUIRED IMPORTED_TARGET libssh>=0.11)
find_package(KF6IdleTime REQUIRED)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets DBus Network Test WebEngineWidgets)
add_executable(lumen-vault-lab main.cpp vault.cpp vault_sync.cpp backups.cpp backup_wizard.cpp bookstack.cpp connection.cpp web_login.cpp vault_web_access.cpp vault_ssh.cpp vault_ssh_access.cpp vault_http.cpp vault_http_access.cpp passkey.cpp vault_passkey.cpp passkey.qrc web_login.qrc icons.qrc)
target_link_libraries(lumen-vault-lab PRIVATE KF6::IdleTime PkgConfig::LIBSSH Qt6::WebEngineWidgets Qt6::Widgets Qt6::DBus Qt6::Network)
target_compile_options(lumen-vault-lab PRIVATE -Wall -Wextra -Werror)
add_executable(vault-ui-test test.cpp vault.cpp vault_sync.cpp backups.cpp backup_wizard.cpp bookstack.cpp connection.cpp web_login.cpp vault_web_access.cpp vault_ssh.cpp vault_ssh_access.cpp vault_http.cpp vault_http_access.cpp passkey.cpp vault_passkey.cpp passkey.qrc web_login.qrc icons.qrc)
target_link_libraries(vault-ui-test PRIVATE KF6::IdleTime PkgConfig::LIBSSH Qt6::WebEngineWidgets Qt6::Widgets Qt6::DBus Qt6::Network Qt6::Test)
target_compile_options(vault-ui-test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME vault-ui COMMAND vault-ui-test)
set_tests_properties(vault-ui PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
