cmake_minimum_required(VERSION 3.16)
project(LumenAppMenu LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(KF6CoreAddons REQUIRED)
find_package(Plasma 6.8 REQUIRED)
add_library(org.lumen.appmenu MODULE factory.cpp)
target_link_libraries(org.lumen.appmenu PRIVATE Qt6::Core Qt6::Widgets KF6::CoreAddons Plasma::Plasma)
set_target_properties(org.lumen.appmenu PROPERTIES PREFIX "")
